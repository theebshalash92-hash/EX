// دالة تشغيل البحث اليدوي عند الضغط على الزر function triggerLookup() { const barcodeInput = document.getElementById('itemBarcode'); if (!barcodeInput) { alert("حقل الباركود غير موجود!"); return; } const barcode = barcodeInput.value.trim(); if (!barcode) { alert("يرجى إدخال أو مسح رقم الباركود أولاً!"); barcodeInput.focus(); return; } lookupBarcodeLocally(barcode); } // ربط زر البحث والضغط على مفتاح Enter تلقائياً document.addEventListener("DOMContentLoaded", () => { const barcodeInput = document.getElementById('itemBarcode'); if (barcodeInput) { barcodeInput.addEventListener("keypress", function(event) { if (event.key === "Enter") { event.preventDefault(); triggerLookup(); } }); } });
